set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(QRMFORGE_BUILD_GUI "Build the standalone JUCE desktop app" ON)
option(QRMFORGE_ENABLE_RNNOISE "Build bundled RNNoise 0.2" ON)
option(QRMFORGE_ENABLE_SPEEXDSP "Use the native SpeexDSP runtime" ON)
option(QRMFORGE_ENABLE_DEEPFILTER "Use the bundled official x86-64 DeepFilterNet backend in an isolated helper" ON)
option(QRMFORGE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(QRMFORGE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
set(nr_sources src/dsp/wiener.cpp src/dsp/speex.cpp src/dsp/deep_filter.cpp)
add_library(standalone_dsp STATIC src/dsp/core.cpp src/dsp/agc.cpp src/dsp/neural.cpp ${nr_sources})
target_include_directories(standalone_dsp PUBLIC src)
target_link_libraries(standalone_dsp PUBLIC Threads::Threads)
target_compile_options(standalone_dsp PRIVATE -Wall -Wextra -Wpedantic)
if(QRMFORGE_ENABLE_RNNOISE)
  include(cmake/RNNoise.cmake)
  target_compile_definitions(standalone_dsp PUBLIC QRMFORGE_ENABLE_RNNOISE=1)
  target_link_libraries(standalone_dsp PUBLIC qrmforge_rnnoise)
endif()
if(QRMFORGE_ENABLE_SPEEXDSP)
  find_library(SPEEXDSP_LIBRARY NAMES speexdsp libspeexdsp.so.1)
  if(SPEEXDSP_LIBRARY)
    target_compile_definitions(standalone_dsp PUBLIC QRMFORGE_ENABLE_SPEEXDSP=1)
    target_link_libraries(standalone_dsp PUBLIC ${SPEEXDSP_LIBRARY})
  else()
    message(WARNING "SpeexDSP runtime unavailable; its mode will be disabled")
  endif()
endif()
set(bundled_deep "${PROJECT_SOURCE_DIR}/third_party/deepfilter/libdeep_filter_ladspa-0.5.6-x86_64-linux-gnu.so")
if(QRMFORGE_ENABLE_DEEPFILTER AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  file(SHA256 "${bundled_deep}" deep_hash)
  if(NOT deep_hash STREQUAL "2ca3205c2911d389604a826a240e745597d50252b5cab81c8248252b335e2236")
    message(FATAL_ERROR "Bundled DeepFilterNet backend checksum mismatch")
  endif()
  add_executable(qrmforge_deepfilter_worker src/dsp/deep_worker.cpp)
  set_target_properties(qrmforge_deepfilter_worker PROPERTIES OUTPUT_NAME qrmforge-deepfilter-worker)
  target_include_directories(qrmforge_deepfilter_worker PRIVATE src third_party/deepfilter)
  target_link_libraries(qrmforge_deepfilter_worker PRIVATE ${CMAKE_DL_LIBS})
  target_sources(standalone_dsp PRIVATE src/dsp/deep_bridge.cpp)
  add_dependencies(standalone_dsp qrmforge_deepfilter_worker)
  foreach(kind LIBDIR LIBEXECDIR)
    if(IS_ABSOLUTE "${CMAKE_INSTALL_${kind}}")
      set(deep_${kind} "${CMAKE_INSTALL_${kind}}")
    else()
      file(RELATIVE_PATH deep_${kind} "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_${kind}}")
    endif()
  endforeach()
  target_compile_definitions(standalone_dsp PUBLIC QRMFORGE_ENABLE_DEEPFILTER=1
    QRMFORGE_DEEPFILTER_INSTALLED_BACKEND="${deep_LIBDIR}/qrmforge/libdeep_filter_ladspa.so"
    QRMFORGE_DEEPFILTER_INSTALLED_HELPER="${deep_LIBEXECDIR}/qrmforge/qrmforge-deepfilter-worker"
    QRMFORGE_DEEPFILTER_BACKEND="${bundled_deep}" QRMFORGE_DEEPFILTER_HELPER="$<TARGET_FILE:qrmforge_deepfilter_worker>")
  file(GENERATE OUTPUT "${PROJECT_BINARY_DIR}/qrmforge-deepfilter-resource.txt"
    CONTENT "$<TARGET_FILE:qrmforge_deepfilter_worker>\n${bundled_deep}\n2ca3205c2911d389604a826a240e745597d50252b5cab81c8248252b335e2236\n")
  install(TARGETS qrmforge_deepfilter_worker RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/qrmforge)
  install(FILES "${bundled_deep}" DESTINATION ${CMAKE_INSTALL_LIBDIR}/qrmforge RENAME libdeep_filter_ladspa.so)
else()
  file(REMOVE "${PROJECT_BINARY_DIR}/qrmforge-deepfilter-resource.txt")
endif()
find_library(WAVPACK_LIBRARY NAMES wavpack libwavpack.so.1 REQUIRED)

target_include_directories(standalone_model PUBLIC third_party/wavpack)
target_link_libraries(standalone_model PUBLIC ${WAVPACK_LIBRARY})
target_link_libraries(standalone_model PUBLIC standalone_dsp Threads::Threads)
